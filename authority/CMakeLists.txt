cmake_minimum_required(VERSION 3.24)
project(IsaacAuthorityResearch VERSION 0.8.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(IA_HOST_TESTS "Build portable state-model and protocol tests with the host compiler" OFF)
if(IA_HOST_TESTS)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic -Werror)
  endif()
  add_library(authority_portable STATIC native/core.cpp native/state_model.cpp native/link.cpp)
  target_include_directories(authority_portable PUBLIC native third_party)
  add_executable(authority-tests tests/core_tests.cpp)
  target_link_libraries(authority-tests PRIVATE authority_portable)
  add_executable(authority-state-tests tests/state_model_tests.cpp)
  target_link_libraries(authority-state-tests PRIVATE authority_portable)
  target_compile_definitions(authority-state-tests PRIVATE "IA_FIXTURE_PATH=\"${CMAKE_CURRENT_SOURCE_DIR}/fixtures/state/native_capture_room_start.bin\"")
  enable_testing()
  add_test(NAME authority-core COMMAND authority-tests)
  add_test(NAME authority-state COMMAND authority-state-tests)
  return()
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Build with Windows MSVC Win32: the J460 game is x86")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /EHsc /utf-8)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
include_directories(native third_party)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/lua/probe.lua" LUA_PROBE_SOURCE)
string(LENGTH "${LUA_PROBE_SOURCE}" LUA_PROBE_LENGTH)
set(LUA_PROBE_LITERALS "")
set(LUA_PROBE_OFFSET 0)
while(LUA_PROBE_OFFSET LESS LUA_PROBE_LENGTH)
  string(SUBSTRING "${LUA_PROBE_SOURCE}" ${LUA_PROBE_OFFSET} 2048 LUA_PROBE_CHUNK)
  string(APPEND LUA_PROBE_LITERALS "R\"IA_LUA(${LUA_PROBE_CHUNK})IA_LUA\"\n")
  math(EXPR LUA_PROBE_OFFSET "${LUA_PROBE_OFFSET} + 2048")
endwhile()
configure_file(native/bootstrap.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/bootstrap.hpp" @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS research/resource_catalog.json)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/research/resource_catalog.json" RESOURCE_CATALOG_SOURCE)
string(LENGTH "${RESOURCE_CATALOG_SOURCE}" RESOURCE_CATALOG_LENGTH)
set(RESOURCE_CATALOG_LITERALS "")
set(RESOURCE_CATALOG_OFFSET 0)
while(RESOURCE_CATALOG_OFFSET LESS RESOURCE_CATALOG_LENGTH)
  string(SUBSTRING "${RESOURCE_CATALOG_SOURCE}" ${RESOURCE_CATALOG_OFFSET} 2048 RESOURCE_CATALOG_CHUNK)
  string(APPEND RESOURCE_CATALOG_LITERALS "R\"IA_ASSETS(${RESOURCE_CATALOG_CHUNK})IA_ASSETS\"\n")
  math(EXPR RESOURCE_CATALOG_OFFSET "${RESOURCE_CATALOG_OFFSET} + 2048")
endwhile()
configure_file(native/resource_catalog.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/resource_catalog.hpp" @ONLY)
add_library(authority_core STATIC native/core.cpp native/platform.cpp native/steam.cpp)
target_link_libraries(authority_core PUBLIC bcrypt)
add_library(authority_state STATIC native/state_model.cpp native/link.cpp)
target_include_directories(authority_state PUBLIC native)
add_library(minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c
  third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde32.c)
target_include_directories(minhook PUBLIC third_party/minhook/include)
set_target_properties(minhook PROPERTIES COMPILE_OPTIONS "/W0;/utf-8")
add_library(authority_lua_bridge STATIC native/lua_bridge.cpp native/mod_runtime.cpp native/mod_runtime_slice2.cpp native/mod_runtime_slice3.cpp native/runtime_suite.cpp native/engine_api.cpp native/resources.cpp native/ui_backend.cpp)
target_include_directories(authority_lua_bridge PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(authority_lua_bridge PUBLIC authority_core)
add_library(isaac_authority_probe SHARED native/probe.cpp)
target_include_directories(isaac_authority_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(isaac_authority_probe PRIVATE authority_core minhook authority_lua_bridge authority_state)
add_executable(isaac-authority native/launcher.cpp)
target_link_libraries(isaac-authority PRIVATE authority_lua_bridge shell32)
add_executable(authority-tests tests/core_tests.cpp)
target_link_libraries(authority-tests PRIVATE authority_core)
add_executable(authority-state-tests tests/state_model_tests.cpp)
target_link_libraries(authority-state-tests PRIVATE authority_state)
target_compile_definitions(authority-state-tests PRIVATE "IA_FIXTURE_PATH=\"${CMAKE_CURRENT_SOURCE_DIR}/fixtures/state/native_capture_room_start.bin\"")
add_executable(authority-bridge-tests tests/bridge_tests.cpp)
target_link_libraries(authority-bridge-tests PRIVATE authority_lua_bridge)
add_executable(authority-runtime-tests tests/runtime_tests.cpp)
target_link_libraries(authority-runtime-tests PRIVATE authority_lua_bridge)
add_executable(authority-resource-tests tests/resource_tests.cpp)
target_link_libraries(authority-resource-tests PRIVATE authority_lua_bridge)
add_executable(authority-suite-tests tests/suite_tests.cpp)
target_link_libraries(authority-suite-tests PRIVATE authority_lua_bridge)
enable_testing()
add_test(NAME authority-core COMMAND authority-tests)
add_test(NAME authority-state COMMAND authority-state-tests)
